    $("progress-label").textContent = `${percent}%`;
    $("progress-fill").style.width = `${percent}%`;
    document.querySelector(".progress").setAttribute("aria-valuenow",String(percent));
    $("next-button").hidden = index === 6;
    $("submit-button").hidden = index !== 6;
    $("survey-error").textContent = "";
  }
  function calculate() {
    const scores = Array(5).fill(0);
    answers.forEach((answer,q) => points[q][answer].forEach((value,genre) => { scores[genre] += value; }));
    const highest = Math.max(...scores);
    const tied = genres.filter((_,i) => scores[i] === highest);
    let primary = tied[0];
    for (let q = 0; q < answers.length && tied.length > 1; q++) {
      const weights = tied.map(genre => points[q][answers[q]][genres.indexOf(genre)]);
      const best = Math.max(...weights);
      if (weights.filter(value => value === best).length === 1) { primary = tied[weights.indexOf(best)]; break; }
    }
    return {primary,tied};
  }
  function renderResult() {
    const result = calculate();
    lastResult = result;
    const mood = moods[answers[1]], sound = sounds[answers[3]];
    $("result-genre").textContent = result.tied.join(" × ");
    $("result-description").textContent = result.tied.length > 1
      ? `${result.tied.join("와 ")}의 매력이 함께 빛나요. ${mood} 분위기를 찾고 ${sound}에 끌리는 당신에게, ${profiles[result.primary].description}`
      : `${mood} 분위기를 찾고 ${sound}에 끌리는 당신. ${profiles[result.primary].description}`;
    const values = [...new Set([
      ...profiles[result.primary].keywords.slice(0,3),
      ...result.tied.filter(g => g !== result.primary).map(g => profiles[g].keywords[0]),
      mood
    ])].slice(0,5);
    $("result-keywords").replaceChildren(...values.map(value => {
      const tag = document.createElement("span");
      tag.className = "keyword";
      tag.textContent = value;
      return tag;
    }));
    document.querySelectorAll(".genre-panel").forEach(panel => { panel.hidden = panel.dataset.genre !== result.primary; });
    $("result-notice").textContent = "";
    showView($("result-view"));
  }

    if (current === 0) showView($("start-view"));
    else showQuestion(current - 1);
  });

    if (event.target.matches('input[type="radio"]')) {
      answers[Number(event.target.name.slice(1))] = Number(event.target.value);
      $("survey-error").textContent = "";
    }
  });
  $("survey-form").addEventListener("submit",event => {
    event.preventDefault();
    if (answers[current] === null) {
      $("survey-error").textContent = "선택지를 하나 골라 주세요.";
      panels[current].querySelector("input").focus();
      return;
    }
    if (current < 6) { showQuestion(current + 1); return; }
    const missing = answers.indexOf(null);
    if (missing !== -1) {
      showQuestion(missing);
      $("survey-error").textContent = "결과를 보기 전에 이 질문에 답해 주세요.";
      panels[missing].querySelector("input").focus();
      return;
    }
    const sameChoice = answers.every(answer => answer === answers[0]);
    if (sameChoice) {
      showView($("pattern-view"));
      return;
    }
    renderResult();
  });

    }
  });
  showQuestion(0);
})();
